cmake_minimum_required(VERSION 3.22.1)
project(lensfun_correction_probe LANGUAGES CXX)
set(PKG_CONFIG_ARGN --static)
set(PKG_CONFIG_USE_CMAKE_PREFIX_PATH FALSE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LENSFUN REQUIRED IMPORTED_TARGET lensfun)
if(ANDROID)
  if(NOT LENSFUN_ARCHIVE)
    message(FATAL_ERROR "LENSFUN_ARCHIVE must name the ARM64 PIC static archive")
  endif()
  add_library(lensfun_link_probe SHARED correction_probe.cpp)
  target_compile_definitions(lensfun_link_probe PRIVATE LENSFUN_ANDROID_LINK_PROBE=1)
  target_link_options(lensfun_link_probe PRIVATE -Wl,--no-undefined)
  target_link_libraries(lensfun_link_probe PRIVATE
    "-Wl,--whole-archive,${LENSFUN_ARCHIVE},--no-whole-archive" PkgConfig::LENSFUN)
else()
  enable_testing()
  add_executable(lensfun_correction_test correction_probe.cpp)
  target_link_libraries(lensfun_correction_test PRIVATE PkgConfig::LENSFUN)
  add_test(NAME lensfun_correction_test COMMAND lensfun_correction_test)
endif()
